cmake_minimum_required(VERSION 3.16)
project(psp3ds_host C)
set(CMAKE_C_STANDARD 11)
add_library(psp_core STATIC source/psp_core.c source/psp_loader.c source/psp_hle.c source/psp_imports.c)
target_include_directories(psp_core PUBLIC include)
enable_testing()
foreach(name core loader integer allegrex4 exceptions hle imports)
  add_executable(test_${name} tests/test_${name}.c)
  target_link_libraries(test_${name} PRIVATE psp_core)
  add_test(NAME ${name} COMMAND test_${name})
endforeach()
